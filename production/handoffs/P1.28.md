```bash
# 1. Python test suite (includes tests/test_prompts.py)
python -m pytest tests/test_prompts.py -v

# 2. Portable few-shot validator run from repo root and an arbitrary directory
python production/handoffs/media/P1.05_check_fewshots.py
(cd /tmp && python -c 'import subprocess, pathlib; subprocess.run(["python", str(pathlib.Path("production/handoffs/media/P1.05_check_fewshots.py").resolve())], check=True)')

# 3. Clean diff and whitespace checks
git diff --check main...HEAD

# 4. UTF-8 transcript verification
python -c '
with open("production/handoffs/media/P1.05_transcript.txt", "r", encoding="utf-8") as f:
    text = f.read()
assert "café" in text, "Missing café in transcript"
print("Transcript: valid UTF-8, length:", len(text))
'

# 5. Measure fixed system prompts and loaded prompt budget with llama-tokenize
# (Set LEILA_LLAMA_BIN and LEILA_MODEL to local artifact locations if not at repo root)
TOKENIZE_BIN="${LEILA_LLAMA_BIN:-tools/llama/llama-b12024}/llama-tokenize"
MODEL_PATH="${LEILA_MODEL:-models/Qwen3.5-4B-Q4_K_M.gguf}"

cd game/data/prompts
for st in 0 1; do
  { cat persona.md; printf 'The player is a boy called {player_name}.\n'; cat stage_$st.md rules.md glosses.md; } > /tmp/sys_s$st.txt
  "$TOKENIZE_BIN" -m "$MODEL_PATH" -f /tmp/sys_s$st.txt | grep -cE '^\s*[0-9]+ -> '
done

cat << 'EOF' > /tmp/loaded_layers.txt
SCENE: A quiet café on a rainy afternoon.
SCENE GOAL: Talk with {player_name}.
Memories of {player_name}:
- likes rainy days
- gave me a music box
- was with Sarah today
TRIGGER: The player mentioned another girl (Sarah) for the 2nd time.
EOF
{ cat persona.md; printf 'The player is a boy called {player_name}.\n'; cat stage_1.md /tmp/loaded_layers.txt rules.md glosses.md; } > /tmp/sys_s1_loaded.txt
"$TOKENIZE_BIN" -m "$MODEL_PATH" -f /tmp/sys_s1_loaded.txt | grep -cE '^\s*[0-9]+ -> '
```
